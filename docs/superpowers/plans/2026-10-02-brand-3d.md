# Brand 3D (logo, welcome intro, login scene) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A live three.js "pasta world" (semolina + the plant's pasta shapes around a simple gold 3D logo) that plays a ~4 s skippable welcome intro and then stays behind a navy login page, plus baked 3D-logo images for the sidebar, mobile header and browser tab.

**Architecture:** Pure, unit-tested modules (maths, colour, SVG parsing, framing, quality tiers, the intro timeline, piece/grain motion, a tiny external store) drive a lazily-loaded three.js world (`world.ts` chunk). One `BrandStage` canvas, mounted once in `App.tsx`, renders while the intro plays (driven by `AppSplash`) or while `LoginPage` is mounted; the login registers DOM "slots" the 3D logo is placed over. A dev-only bake page renders the static logo images and fallback posters from the same code.

**Tech Stack:** React 18 + TypeScript + Vite 5, three.js 0.186 (EffectComposer/UnrealBloomPass/OutputPass, RoomEnvironment, BufferGeometryUtils), Framer Motion, Vitest (node env), Node 25 + Chrome DevTools Protocol for bake/stills scripts.

**Spec:** `docs/superpowers/specs/2026-10-02-brand-3d-design.md`

Two refinements of the spec, decided while planning:
- The spec's `quality.ts` held both tiers and slot maths; the plan splits the camera geometry into `framing.ts` (`fitDistance`, `placeInSlot`) and keeps tiers in `quality.ts`.
- The spec said the main bundle does not grow. The small pure modules the splash and login import (store, timeline, maths) do land in the main chunk (a few KB); `three` and the scene stay in the lazy chunk. Task 15 checks the main chunk grows by under 10 KB gzipped.

```ts
/**
 * Maths shared by the brand 3D scene, ported from the brand-motion-design
 * skill's kit (kit/film.js). Pure: no DOM, no WebGL, so it runs under Vitest.
 */

export type Vec2 = [number, number];
export type Vec3 = [number, number, number];

export const clamp01 = (u: number): number => (u < 0 ? 0 : u > 1 ? 1 : u);
export const lerp = (a: number, b: number, u: number): number => a + (b - a) * u;
/** How far through [start, start + length] time `t` is, clamped to 0..1. */
export const progress = (t: number, start: number, length: number): number => clamp01((t - start) / length);

/** A CSS cubic-bezier() as a function of 0..1 (the kit's PF.bezier). */
export function bezier(x1: number, y1: number, x2: number, y2: number): (x: number) => number {
  const cx = 3 * x1;
  const bx = 3 * (x2 - x1) - cx;
  const ax = 1 - cx - bx;
  const cy = 3 * y1;
  const by = 3 * (y2 - y1) - cy;
  const ay = 1 - cy - by;
  const sx = (u: number) => ((ax * u + bx) * u + cx) * u;
  const sy = (u: number) => ((ay * u + by) * u + cy) * u;
  const dx = (u: number) => (3 * ax * u + 2 * bx) * u + cx;
  function solve(x: number): number {
    let u = x;
    for (let i = 0; i < 8; i++) {
      const e = sx(u) - x;
      if (Math.abs(e) < 1e-6) return u;
      const d = dx(u);
      if (Math.abs(d) < 1e-6) break;
      u -= e / d;
    }
    let lo = 0;
    let hi = 1;
    u = x;
    while (lo < hi) {
      const v = sx(u);
      if (Math.abs(v - x) < 1e-6) return u;
      if (x > v) lo = u;
      else hi = u;
      u = (lo + hi) / 2;
      if (hi - lo < 1e-7) break;
    }
    return u;
  }
  return (x) => (x <= 0 ? 0 : x >= 1 ? 1 : sy(solve(x)));
}

/** The app's motion token: EASE = [0.22, 1, 0.36, 1] in LoginPage.tsx and BrandLoader.tsx. */
export const brandEase = bezier(0.22, 1, 0.36, 1);
/** CSS `ease`: the timing of logo-mark.svg's own build. */
export const cssEase = bezier(0.25, 0.1, 0.25, 1);
/** Symmetric ease for camera moves and the skip fast-forward. */
export const easeInOut = bezier(0.65, 0, 0.35, 1);

/** Seeded PRNG (mulberry32), the scene's only randomness: every layout repeats exactly. */
export function random(seed: number): () => number {
  let s = seed >>> 0;
  return () => {
    s = (s + 0x6d2b79f5) >>> 0;
    let r = Math.imul(s ^ (s >>> 15), 1 | s);
    r = (r + Math.imul(r ^ (r >>> 7), 61 | r)) ^ r;
    return ((r ^ (r >>> 14)) >>> 0) / 4294967296;
  };
}
```

```ts
/**
 * Brand colour helpers. `untone` is the fix measured in the skill's BRAND.md:
 * three's ACES tone mapping shifts a scene background off its swatch, so the
 * background and fog use the colour ACES maps back onto #071F43 exactly.
 * Pure, so the round trip is tested without WebGL.
 */
import { clamp01, lerp, type Vec3 } from "./math";

/** "#F5A900" -> [r, g, b] in 0..1, sRGB as the brand tokens are written. */
export function hexToRgb(hex: string): Vec3 {
  const h = hex.replace("#", "");
  const full = h.length === 3 ? h.split("").map((c) => c + c).join("") : h;
  return [0, 2, 4].map((i) => parseInt(full.slice(i, i + 2), 16) / 255) as Vec3;
}

export const srgbToLinear = (c: number): number => (c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4));
export const linearToSrgb = (c: number): number => (c <= 0.0031308 ? c * 12.92 : 1.055 * Math.pow(c, 1 / 2.4) - 0.055);

/** Mix two brand hexes (the kit's PF.mix): u = 0 gives `a`, 1 gives `b`. */
export function mixHex(a: string, b: string, u: number): string {
  const x = hexToRgb(a);
  const y = hexToRgb(b);
  return "#" + x.map((v, i) => Math.round(lerp(v, y[i], clamp01(u)) * 255).toString(16).padStart(2, "0")).join("");
}

// three.js ACESFilmicToneMapping (tonemapping_pars_fragment), written row-major.
const ACES_IN = [0.59719, 0.35458, 0.04823, 0.076, 0.90834, 0.01566, 0.0284, 0.13383, 0.83777];
const ACES_OUT = [1.60475, -0.53108, -0.07367, -0.10208, 1.10813, -0.00605, -0.00327, -0.07276, 1.07602];

function mul(m: readonly number[], v: Vec3): Vec3 {
  return [
    m[0] * v[0] + m[1] * v[1] + m[2] * v[2],
    m[3] * v[0] + m[4] * v[1] + m[5] * v[2],
    m[6] * v[0] + m[7] * v[1] + m[8] * v[2],
  ];
}

function inverse(m: readonly number[]): number[] {
  const [a, b, c, d, e, f, g, h, i] = m;
  const A = e * i - f * h;
  const B = -(d * i - f * g);
  const C = d * h - e * g;
  const det = a * A + b * B + c * C;
  return [
    A / det, -(b * i - c * h) / det, (b * f - c * e) / det,
    B / det, (a * i - c * g) / det, -(a * f - c * d) / det,
    C / det, -(a * h - b * g) / det, (a * e - b * d) / det,
  ];
}

const fit = (x: number) => (x * (x + 0.0245786) - 0.000090537) / (x * (0.983729 * x + 0.432951) + 0.238081);

function unfit(y: number): number {
  const a = 1 - 0.983729 * y;
  const b = 0.0245786 - 0.432951 * y;
  const c = -0.000090537 - 0.238081 * y;
  return (-b + Math.sqrt(b * b - 4 * a * c)) / (2 * a);
}

/** What three's ACES filmic tone mapping at `exposure` does to a linear colour (linear out, before the sRGB encode). */
export function acesFilmic(linear: Vec3, exposure: number): Vec3 {
  const v = mul(ACES_IN, linear.map((c) => (c * exposure) / 0.6) as Vec3);
  return mul(ACES_OUT, v.map(fit) as Vec3).map(clamp01) as Vec3;
}

/** The linear colour that ACES at `exposure` turns back into exactly `hex`. */
export function untone(hex: string, exposure: number): Vec3 {
  const target = hexToRgb(hex).map(srgbToLinear) as Vec3;
  const v = mul(inverse(ACES_OUT), target).map(unfit) as Vec3;
  return mul(inverse(ACES_IN), v).map((c) => (c * 0.6) / exposure) as Vec3;
}
```

```ts
/**
 * Reads the logo's geometry straight from logo-mark.svg, so the 3D mark is
 * built from the same file the app shows in 2D and follows it if it changes.
 * Pure string parsing (no DOMParser), so it runs under Vitest. It handles
 * exactly what the mark uses (M/m, L/l, C/c, Z/z) and refuses anything else.
 */
import type { Vec2 } from "./math";

export type Segment = { kind: "L"; to: Vec2 } | { kind: "C"; c1: Vec2; c2: Vec2; to: Vec2 };

export interface SubPath {
  start: Vec2;
  segments: Segment[];
  closed: boolean;
}

export interface GlyphStroke {
  width: number;
  subpaths: SubPath[];
}

export interface LogoSpec {
  viewBox: number;
  ring: { cx: number; cy: number; r: number; width: number };
  glyph: { translate: Vec2; scale: number; strokes: GlyphStroke[] };
}

const attrs = (tag: string): Record<string, string> =>
  Object.fromEntries([...tag.matchAll(/([\w:-]+)="([^"]*)"/g)].map((m) => [m[1], m[2]]));

export function parsePathData(d: string): SubPath[] {
  const tokens = d.match(/[A-Za-z]|-?(?:\d+\.?\d*|\.\d+)(?:[eE][-+]?\d+)?/g) ?? [];
  const out: SubPath[] = [];
  let i = 0;
  let cmd = "";
  let cur: Vec2 = [0, 0];
  let path: SubPath | null = null;
  const num = (): number => {
    const v = Number(tokens[i++]);
    if (Number.isNaN(v)) throw new Error(`logo path: expected a number in "${d}"`);
    return v;
  };
  // Relative points (lower-case commands) are offsets from the point the segment starts at.
  const point = (relative: boolean): Vec2 => {
    const x = num();
    const y = num();
    return relative ? [cur[0] + x, cur[1] + y] : [x, y];
  };
  const current = (): SubPath => {
    if (!path) throw new Error(`logo path: drawing before a move in "${d}"`);
    return path;
  };

  while (i < tokens.length) {
    if (/[A-Za-z]/.test(tokens[i])) cmd = tokens[i++];
    else if (!cmd) throw new Error(`logo path: number before any command in "${d}"`);
    const relative = cmd === cmd.toLowerCase();
    switch (cmd.toUpperCase()) {
      case "M":
        cur = point(relative);
        path = { start: cur, segments: [], closed: false };
        out.push(path);
        cmd = relative ? "l" : "L"; // further pairs after a move are line-tos
        break;
      case "L": {
        const to = point(relative);
        current().segments.push({ kind: "L", to });
        cur = to;
        break;
      }
      case "C": {
        const c1 = point(relative);
        const c2 = point(relative);
        const to = point(relative);
        current().segments.push({ kind: "C", c1, c2, to });
        cur = to;
        break;
      }
      case "Z":
        current().closed = true;
        cur = current().start;
        cmd = "";
        break;
      default:
        throw new Error(`logo path: unsupported command "${cmd}" in "${d}"`);
    }
  }
  return out;
}

export function parseLogoSvg(svg: string): LogoSpec {
  const viewBox = Number(/viewBox="0 0 ([\d.]+) [\d.]+"/.exec(svg)?.[1]);
  const circles = [...svg.matchAll(/<circle\b[^>]*>/g)].map((m) => attrs(m[0]));
  const ring = circles.find((c) => (c.class ?? "").includes("orbit")) ?? circles[0];
  const glyph = [...svg.matchAll(/<g\b([^>]*)>([\s\S]*?)<\/g>/g)].find((g) => (attrs(g[1]).class ?? "").includes("glyph"));
  if (!viewBox || !ring || !glyph) throw new Error("logo-mark.svg: viewBox, ring or glyph group not found");
  const transform = /translate\(([-\d.]+)[ ,]+([-\d.]+)\)\s*scale\(([-\d.]+)\)/.exec(attrs(glyph[1]).transform ?? "");
  if (!transform) throw new Error("logo-mark.svg: glyph transform is not translate(...) scale(...)");
  const strokes = [...glyph[2].matchAll(/<path\b[^>]*>/g)].map((m) => {
    const a = attrs(m[0]);
    return { width: Number(a["stroke-width"]), subpaths: parsePathData(a.d) };
  });
  return {
    viewBox,
    ring: { cx: Number(ring.cx), cy: Number(ring.cy), r: Number(ring.r), width: Number(ring["stroke-width"]) },
    glyph: { translate: [Number(transform[1]), Number(transform[2])], scale: Number(transform[3]), strokes },
  };
}

/** A glyph point (path units) in logo space: centred, y up, the viewBox mapped to [-1, 1]. */
export function toLogoSpace([x, y]: Vec2, spec: LogoSpec): Vec2 {
  const half = spec.viewBox / 2;
  const [tx, ty] = spec.glyph.translate;
  return [(tx + spec.glyph.scale * x - half) / half, -(ty + spec.glyph.scale * y - half) / half];
}
```

```ts
/**
 * Camera framing for the brand scene. Pure. `fitDistance` keeps a composition
 * on screen at any aspect (the kit's 9:16 "narrow slice" trap); `placeInSlot`
 * puts the 3D logo exactly over a DOM box (the skill's "measure, never guess").
 */

export interface Rect {
  left: number;
  top: number;
  width: number;
  height: number;
}

export interface SlotPlacement {
  x: number;
  y: number;
  z: number;
  /** World height the box covers at that depth. */
  size: number;
}

/** The logo sits this far in front of the camera in a login slot: nearer than any pasta (see motion.test.ts). */
export const LOGO_SLOT_DEPTH = 3;

const halfTan = (fovDeg: number) => Math.tan((fovDeg * Math.PI) / 360);

/** Camera distance that shows a sphere of `radius` around the target on the screen's shorter axis (vertical fov). */
export function fitDistance(radius: number, fovDeg: number, aspect: number): number {
  return radius / (halfTan(fovDeg) * Math.min(1, aspect));
}

/** Camera-space position, and world height, for an object `depth` in front of the camera to fill `rect` (CSS px) of a `vw` × `vh` viewport. */
export function placeInSlot(rect: Rect, vw: number, vh: number, fovDeg: number, depth: number): SlotPlacement {
  const halfH = depth * halfTan(fovDeg);
  const halfW = halfH * (vw / vh);
  const ndcX = ((rect.left + rect.width / 2) / vw) * 2 - 1;
  const ndcY = 1 - ((rect.top + rect.height / 2) / vh) * 2;
  return { x: ndcX * halfW, y: ndcY * halfH, z: -depth, size: (Math.min(rect.width, rect.height) / vh) * 2 * halfH };
}
```

```ts
/**
 * Quality tiers for the live scene, chosen by measuring (spec §5): start high,
 * step down one tier when the median frame is too slow, never step back up.
 */

export type Tier = "high" | "low" | "still";

export interface TierSettings {
  grains: number;
  pieces: number;
  bloom: boolean;
  maxPixelRatio: number;
}

export const TIERS: Record<"high" | "low", TierSettings> = {
  high: { grains: 1200, pieces: 60, bloom: true, maxPixelRatio: 1.5 },
  low: { grains: 500, pieces: 30, bloom: false, maxPixelRatio: 1 },
};

export const SAMPLE_FRAMES = 60;
export const SLOW_MS = 22;
export const STILL_MS = 33;

export function median(xs: readonly number[]): number {
  const sorted = [...xs].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  return sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
}

/**
 * The tier after a full sample of frame times. One step down when the median is
 * too slow, never up. The intro never freezes (a frozen intro reads as broken),
 * so `allowStill` is false while it plays.
 */
export function nextTier(current: Tier, frameMs: readonly number[], allowStill: boolean): Tier {
  if (current === "still" || frameMs.length < SAMPLE_FRAMES) return current;
  const m = median(frameMs);
  if (current === "high") return m > SLOW_MS ? "low" : "high";
  return allowStill && m > STILL_MS ? "still" : "low";
}

export function pixelRatio(device: number, tier: Tier): number {
  return Math.min(device, tier === "high" ? TIERS.high.maxPixelRatio : TIERS.low.maxPixelRatio);
}
```

```ts
/**
 * The welcome intro's choreography and the camera, as pure functions of time:
 * the skill's rule that every frame is a function of t, nothing carried from
 * one frame to the next. Beats from the spec, §2.
 */
import { fitDistance } from "./framing";
import { brandEase, cssEase, easeInOut, lerp, progress, type Vec3 } from "./math";

/** Beat times in seconds from the intro's first rendered frame. */
export const INTRO = {
  semolinaEnd: 1.2,
  pastaStart: 1.2,
  pastaEnd: 2.4,
  orbitStart: 1.8,
  orbitLength: 1.2,
  ringStart: 2.4,
  ringEnd: 3.1,
  glyphStart: 2.8,
  glyphEnd: 3.3,
  wordmarkIn: 3.0,
  handoff: 3.6,
  end: 4.0,
  /** How long a skip takes to fast-forward to the handoff. */
  skip: 0.3,
} as const;

export interface IntroState {
  /** 0..1: the semolina stream flowing in from the depth. */
  stream: number;
  /** 0..1: grains condensing into pasta (each grain staggers inside this, motion.ts). */
  condense: number;
  /** 0..1: pasta tumbling out from where it formed to its track. */
  orbit: number;
  /** 0..1: opacity of the HTML wordmark and tagline. */
  wordmark: number;
  /** 0..1: camera pull-back and the logo's flight to its login slot. */
  handoff: number;
}

export function introState(t: number): IntroState {
  return {
    stream: progress(t, 0, INTRO.semolinaEnd),
    condense: progress(t, INTRO.pastaStart, INTRO.pastaEnd - INTRO.pastaStart),
    orbit: brandEase(progress(t, INTRO.orbitStart, INTRO.orbitLength)),
    wordmark: brandEase(progress(t, INTRO.wordmarkIn, 0.3)) * (1 - progress(t, INTRO.handoff, 0.2)),
    handoff: easeInOut(progress(t, INTRO.handoff, INTRO.end - INTRO.handoff)),
  };
}

export interface LogoState {
  /** 0..1: how much of the ring is drawn, clockwise from the top. */
  ring: number;
  /** 0..1: the wheat glyph rising into the ring. */
  glyph: number;
  /** 0..1: the bloom flash when the ring closes. */
  flash: number;
}

/**
 * The mark's build. During the intro it follows the intro's beats. On a login
 * reached without the intro it plays logo-mark.svg's own CSS build, `build`
 * seconds after the scene appeared (ring 1.6 s from 0.1 s, glyph 0.6 s from
 * 0.5 s, CSS `ease`); Infinity shows the finished mark.
 */
export function logoState(intro: number | null, build: number): LogoState {
  if (intro !== null) {
    const ring = cssEase(progress(intro, INTRO.ringStart, INTRO.ringEnd - INTRO.ringStart));
    const glyph = brandEase(progress(intro, INTRO.glyphStart, INTRO.glyphEnd - INTRO.glyphStart));
    const flash = intro < INTRO.ringEnd ? progress(intro, INTRO.ringEnd - 0.1, 0.1) : 1 - progress(intro, INTRO.ringEnd, 0.35);
    return { ring, glyph, flash };
  }
  return { ring: cssEase(progress(build, 0.1, 1.6)), glyph: cssEase(progress(build, 0.5, 0.6)), flash: 0 };
}

/**
 * Intro time after a skip: from `skippedAt` the clock fast-forwards to the
 * handoff pose in INTRO.skip seconds, then runs on at normal speed. A skip at
 * or after the handoff changes nothing. `since` is real seconds since the skip.
 */
export function skipTime(skippedAt: number, since: number): number {
  if (skippedAt >= INTRO.handoff) return skippedAt + since;
  if (since >= INTRO.skip) return INTRO.handoff + (since - INTRO.skip);
  return lerp(skippedAt, INTRO.handoff, easeInOut(since / INTRO.skip));
}

export const FOV = 35;
/** World radius the intro keeps on the short axis: the logo (radius 0.88) fills 40 % of it. */
export const INTRO_FIT = 2.2;
/** World radius the login keeps on the short axis: wider, so the pasta drifts around the card. */
export const LOGIN_FIT = 3.4;

/** Orbit camera: angles in degrees around `target`. */
export interface CameraPose {
  radius: number;
  azimuth: number;
  elevation: number;
  target: Vec3;
}

function loginPose(time: number, aspect: number): CameraPose {
  return {
    radius: fitDistance(LOGIN_FIT, FOV, aspect),
    azimuth: 4 * Math.sin(time * 0.05),
    elevation: 4 + 1.5 * Math.sin(time * 0.037),
    target: [0, 0, -1],
  };
}

function introPose(t: number, aspect: number): CameraPose {
  // Starts close in the grains and settles back onto the logo by the end of the pasta beat.
  const settle = brandEase(progress(t, 0, INTRO.pastaEnd));
  const drift = progress(t, 0, INTRO.handoff);
  return {
    radius: fitDistance(INTRO_FIT, FOV, aspect) * lerp(0.75, 1, settle),
    azimuth: lerp(-18, -6, drift),
    elevation: lerp(8, 3, drift),
    target: [0, 0, 0],
  };
}

/** The camera at intro time `intro` (null after the intro) and stage time `time`. */
export function cameraPose(intro: number | null, time: number, aspect: number): CameraPose {
  const login = loginPose(time, aspect);
  if (intro === null) return login;
  const u = introState(intro).handoff;
  if (u >= 1) return login;
  const from = introPose(Math.min(intro, INTRO.handoff), aspect);
  return {
    radius: lerp(from.radius, login.radius, u),
    azimuth: lerp(from.azimuth, login.azimuth, u),
    elevation: lerp(from.elevation, login.elevation, u),
    target: [lerp(from.target[0], login.target[0], u), lerp(from.target[1], login.target[1], u), lerp(from.target[2], login.target[2], u)],
  };
}

/** Pointer parallax: up to 2.5° sideways and 2° vertically (pointer in -1..1). */
export function withPointer(pose: CameraPose, pointer: { x: number; y: number }): CameraPose {
  return { ...pose, azimuth: pose.azimuth + pointer.x * 2.5, elevation: pose.elevation - pointer.y * 2 };
}

export function cameraPosition(pose: CameraPose): Vec3 {
  const a = (pose.azimuth * Math.PI) / 180;
  const e = (pose.elevation * Math.PI) / 180;
  const [tx, ty, tz] = pose.target;
  return [tx + pose.radius * Math.cos(e) * Math.sin(a), ty + pose.radius * Math.sin(e), tz + pose.radius * Math.cos(e) * Math.cos(a)];
}
```

```ts
/**
 * Where every pasta piece and semolina grain is at a given time. Pure, so the
 * layout's promises are tested: pasta never covers the 3D logo (spec §1–3).
 *
 * Pieces run on two tracks. "orbit" pieces circle behind the logo's plane
 * (z ≤ -1.4, so even a spaghetti's tip stays behind it). "side" pieces drift
 * up the edges of the frame, nearer the camera but never across the logo.
 */
import { brandEase, clamp01, lerp, progress, random, type Vec3 } from "./math";
import { INTRO, type IntroState } from "./timeline";

export type PastaShape = "spaghetti" | "coquillette" | "torsade" | "vermicelle";
export const SHAPES: readonly PastaShape[] = ["spaghetti", "coquillette", "torsade", "vermicelle"];

/** 20-piece pattern with the plant's mix (4 spaghetti, 7 coquillettes, 6 torsadés, 3 vermicelles); any prefix keeps it. */
const PATTERN: readonly PastaShape[] = [
  "coquillette", "torsade", "spaghetti", "coquillette", "vermicelle",
  "torsade", "coquillette", "spaghetti", "torsade", "coquillette",
  "torsade", "spaghetti", "coquillette", "vermicelle", "torsade",
  "coquillette", "spaghetti", "torsade", "coquillette", "vermicelle",
];

/** Half the longest piece (spaghetti 1.8 long, scaled up to 1.3). */
export const PIECE_EXTENT = 1.2;
const SIDE_SPAN = 14; // side pieces wrap over y in [-7, 7]: off screen at every framing

export interface PieceSeed {
  index: number;
  shape: PastaShape;
  track: "orbit" | "side";
  /** orbit: ellipse semi-axes, centre depth, height, phase (rad), angular speed (rad/s). */
  ax: number;
  az: number;
  centerZ: number;
  height: number;
  phase: number;
  speed: number;
  /** side: fixed x and z, rise speed (units/s); `phase` sets the starting height. */
  x: number;
  z: number;
  rise: number;
  size: number;
  /** rad/s per axis: one dominant axis, a full tumble every 20–40 s. */
  spin: Vec3;
  tilt: Vec3;
  /** 0..1 along the gold ramp. */
  tone: number;
  /** 0..1 stagger of the moment it forms. */
  formDelay: number;
}

export interface Pose {
  position: Vec3;
  rotation: Vec3;
  scale: number;
}

export function pieceSeeds(count: number, seed: number): PieceSeed[] {
  return Array.from({ length: count }, (_, index) => {
    const r = random(seed + index * 7919);
    const spin: Vec3 = [lerp(-0.08, 0.08, r()), lerp(-0.08, 0.08, r()), lerp(-0.08, 0.08, r())];
    spin[Math.floor(r() * 3)] = (r() < 0.5 ? -1 : 1) * lerp(0.16, 0.31, r());
    return {
      index,
      shape: PATTERN[index % PATTERN.length],
      track: r() < 0.7 ? "orbit" : "side",
      ax: lerp(2.0, 5.0, r()),
      az: lerp(0.6, 1.4, r()),
      centerZ: lerp(-2.8, -4.5, r()),
      height: lerp(-1.8, 1.8, r()),
      phase: r() * Math.PI * 2,
      speed: lerp(0.04, 0.09, r()),
      x: (r() < 0.5 ? -1 : 1) * lerp(2.6, 4.2, r()),
      z: lerp(-1, 2, r()),
      rise: lerp(0.05, 0.12, r()),
      size: lerp(0.8, 1.3, r()),
      spin,
      tilt: [r() * Math.PI * 2, r() * Math.PI * 2, r() * Math.PI * 2],
      tone: r(),
      formDelay: r(),
    };
  });
}

function home(p: PieceSeed, time: number): Vec3 {
  if (p.track === "orbit") {
    const a = p.phase + p.speed * time;
    return [p.ax * Math.cos(a), p.height + 0.2 * Math.sin(time * 0.3 + p.phase), p.centerZ + p.az * Math.sin(a)];
  }
  const y = ((((p.phase / (Math.PI * 2)) * SIDE_SPAN + p.rise * time) % SIDE_SPAN) + SIDE_SPAN) % SIDE_SPAN - SIDE_SPAN / 2;
  return [p.x, y, p.z];
}

/**
 * A piece's pose at stage `time`. During the intro it forms out of the grains
 * nearer the centre (x and y pulled in, depth kept, so orbit pieces stay behind
 * the logo) and tumbles out onto its track, arriving exactly on the login pose.
 */
export function piecePose(p: PieceSeed, time: number, intro: IntroState | null): Pose {
  const position = home(p, time);
  const rotation: Vec3 = [p.tilt[0] + p.spin[0] * time, p.tilt[1] + p.spin[1] * time, p.tilt[2] + p.spin[2] * time];
  if (!intro) return { position, rotation, scale: p.size };
  const formed = brandEase(progress(intro.condense, 0.35 + p.formDelay * 0.4, 0.25));
  if (formed >= 1 && intro.orbit >= 1) return { position, rotation, scale: p.size };
  const pull = lerp(0.55, 1, intro.orbit);
  const turn = Math.PI * (1 - intro.orbit); // the last half tumble settles as it reaches its track
  return {
    position: [position[0] * pull, position[1] * pull, position[2]],
    rotation: [rotation[0] + turn, rotation[1], rotation[2] + turn * 0.5],
    scale: p.size * formed,
  };
}

export interface GrainSeed {
  index: number;
  /** The piece it condenses into (modulo the live piece count). */
  piece: number;
  /** 0..0.35: how far behind the stream's head it travels. */
  lag: number;
  angle: number;
  radial: number;
  offset: Vec3;
  dust: Vec3;
  rise: number;
  /** A spark: drawn brighter than white so it alone blooms. */
  glow: boolean;
  tone: number;
}

/** The volume the login's floating dust fills. */
export const DUST = { x: 9, y: 7, zNear: 1.5, zFar: -12 } as const;

export function grainSeeds(count: number, seed: number): GrainSeed[] {
  return Array.from({ length: count }, (_, index) => {
    const r = random(seed + index * 104729);
    return {
      index,
      piece: index,
      lag: r() * 0.35,
      angle: r() * Math.PI * 2,
      radial: r(),
      offset: [lerp(-0.22, 0.22, r()), lerp(-0.22, 0.22, r()), lerp(-0.22, 0.22, r())],
      dust: [lerp(-DUST.x, DUST.x, r()), lerp(-DUST.y, DUST.y, r()), lerp(DUST.zFar, DUST.zNear, r())],
      rise: lerp(0.03, 0.1, r()),
      glow: r() < 0.04,
      tone: r(),
    };
  });
}

/** The semolina stream: a spiral from the depth (v = 0) to the centre (v = 1), the splash art's wave in 3D. */
function streamPoint(g: GrainSeed, v: number, out: Vec3): void {
  const a = g.angle + v * Math.PI * 3;
  const radius = lerp(3.2, 0.5, v) + g.radial * 0.4;
  out[0] = radius * Math.cos(a);
  out[1] = radius * Math.sin(a) * 0.6 + 0.3 * Math.sin(v * 6 + g.angle);
  out[2] = lerp(-14, 0, v);
}

/** Writes a grain's position into `out` and returns its opacity (0..1). */
export function grainPosition(
  g: GrainSeed,
  time: number,
  intro: { t: number; state: IntroState } | null,
  pieces: readonly Pose[],
  pieceCount: number,
  out: Vec3,
): number {
  if (intro && intro.t < INTRO.pastaEnd) {
    const v = clamp01(intro.state.stream * 1.35 - g.lag);
    streamPoint(g, v, out);
    const c = brandEase(progress(intro.state.condense, g.lag, 0.55));
    if (c > 0) {
      const target = pieces[g.piece % pieceCount].position;
      out[0] = lerp(out[0], target[0] + g.offset[0], c);
      out[1] = lerp(out[1], target[1] + g.offset[1], c);
      out[2] = lerp(out[2], target[2] + g.offset[2], c);
    }
    return v > 0 ? lerp(0.3, 1, v) * (1 - progress(c, 0.75, 0.25)) : 0;
  }
  // Dust: the same grains float up through the volume, fading in once the pasta has formed.
  const span = DUST.y * 2;
  out[0] = g.dust[0];
  out[1] = ((((g.dust[1] + DUST.y + g.rise * time) % span) + span) % span) - DUST.y;
  out[2] = g.dust[2];
  const fade = intro ? progress(intro.t, INTRO.pastaEnd, 1) : 1;
  return lerp(0.35, 0.85, g.tone) * fade;
}
```

```ts
/**
 * The brand stage's shared state: which screen wants the 3D, the intro's phase
 * and its clock. One small external store, because the splash (outside the
 * router) and the login page (inside it) drive one canvas together.
 */
import { useSyncExternalStore } from "react";
import { skipTime } from "./timeline";

export type IntroPhase = "none" | "playing" | "handoff" | "done";

export interface StageState {
  introPhase: IntroPhase;
  /** The intro clock runs: from the first live frame, or from the poster timeout. */
  introStarted: boolean;
  /** The login page's logo slots while it is mounted, null otherwise. */
  loginSlots: readonly HTMLElement[] | null;
  /** The live scene has drawn its first frame. */
  ready: boolean;
  /** No live 3D here: reduced motion, no WebGL 2, a lost context or a failed chunk. */
  fallback: boolean;
}

const INITIAL: StageState = { introPhase: "none", introStarted: false, loginSlots: null, ready: false, fallback: false };

export function createStageStore(now: () => number = () => performance.now()) {
  let state = INITIAL;
  const listeners = new Set<() => void>();
  let startMs: number | null = null;
  let skipMs: number | null = null;
  let skippedAt = 0;
  let frozen: number | null = null;

  const set = (patch: Partial<StageState>) => {
    state = { ...state, ...patch };
    listeners.forEach((listener) => listener());
  };

  /** Intro time in seconds: 0 until started; after a skip, the fast-forward (timeline.skipTime). */
  const introTime = (): number => {
    if (frozen !== null) return frozen;
    if (startMs === null) return 0;
    const at = now();
    return skipMs === null ? (at - startMs) / 1000 : skipTime(skippedAt, (at - skipMs) / 1000);
  };

  const startIntro = () => {
    if (startMs !== null) return;
    startMs = now();
    set({ introStarted: true });
  };

  /** A skip before the 3D has started the clock starts it, so the intro still ends at once. */
  const skipIntro = () => {
    if (frozen !== null || skipMs !== null) return;
    startIntro();
    skippedAt = introTime();
    skipMs = now();
  };

  return {
    get: () => state,
    subscribe: (listener: () => void) => {
      listeners.add(listener);
      return () => {
        listeners.delete(listener);
      };
    },
    set,
    introTime,
    startIntro,
    skipIntro,
    /** Dev stills (`?brand-t=`): pin the intro clock at `t` seconds. */
    freezeIntro: (t: number) => {
      frozen = t;
    },
```

```ts
 * the brand-motion-design skill's kit/three.js used here, with its measured
 * traps built in: bloom threshold 1.0 (lower turns the scene to milk),

  CanvasTexture,

import { OutputPass } from "three/addons/postprocessing/OutputPass.js";
import { RenderPass } from "three/addons/postprocessing/RenderPass.js";

export interface Stage {
  renderer: WebGLRenderer;
  environment: Texture;
  render(scene: Scene, camera: Camera): void;
  setSize(width: number, height: number, pixelRatio: number): void;
  setBloom(on: boolean): void;
  dispose(): void;
}

export interface StageOptions {
  /** Transparent background (the logo bake). Bloom stays off: its pass writes alpha 1. */
  transparent?: boolean;
  bloom?: boolean;
  /** Keep the drawing buffer so the bake can read pixels back. */
  preserve?: boolean;
}

export function createStage(canvas: HTMLCanvasElement, { transparent = false, bloom = true, preserve = false }: StageOptions = {}): Stage {
  const renderer = new WebGLRenderer({ canvas, antialias: true, alpha: transparent, preserveDrawingBuffer: preserve, powerPreference: "high-performance" });
  renderer.outputColorSpace = SRGBColorSpace;
  renderer.toneMapping = ACESFilmicToneMapping;
  renderer.toneMappingExposure = EXPOSURE;
  if (transparent) renderer.setClearColor(0x000000, 0);

  const pmrem = new PMREMGenerator(renderer);
  const room = new RoomEnvironment();
  const environment = pmrem.fromScene(room, 0.04).texture;
  room.dispose();
  pmrem.dispose();

  const composer = new EffectComposer(renderer);
  const renderPass = new RenderPass(new Scene(), new PerspectiveCamera());
  const bloomPass = new UnrealBloomPass(new Vector2(1, 1), 0.7, 0.55, 1.0);
  composer.addPass(renderPass);
  composer.addPass(bloomPass);
  composer.addPass(new OutputPass());
  let useBloom = bloom && !transparent;

  return {
    renderer,
    environment,
    render(scene, camera) {
      if (!useBloom) {
        renderer.render(scene, camera);
        return;
      }
      renderPass.scene = scene;
      renderPass.camera = camera;
      composer.render();
    },
    setSize(width, height, pixelRatio) {
      renderer.setPixelRatio(pixelRatio);
      renderer.setSize(width, height, false);
      composer.setPixelRatio(pixelRatio);
      composer.setSize(width, height);
    },
    setBloom(on) {
      useBloom = on && !transparent;
    },
    dispose() {
      bloomPass.dispose();
      composer.dispose();
      environment.dispose();
      renderer.dispose();
      // Release the GPU context now: logout/login cycles must not pile contexts up.
      renderer.forceContextLoss();
    },
  };
}

  texture.colorSpace = SRGBColorSpace;
```

```ts
/**
 * The simple 3D logo: logo-mark.svg as polished gold wire. The ring is a tube
 * that draws itself clockwise from the top (the SVG's own build), each glyph
 * stroke a round-ended tube along its path. Built from the SVG file itself.
 */
import {
  AdditiveBlending,
  Color,
  CubicBezierCurve3,
  Curve,
  CurvePath,
  Group,
  LineCurve3,
  Mesh,
  MeshBasicMaterial,
  PlaneGeometry,
  SphereGeometry,
  TubeGeometry,
  Vector3,
  type BufferGeometry,
  type Material,
} from "three";
import { mergeGeometries } from "three/addons/utils/BufferGeometryUtils.js";
import logoSvg from "../../../assets/animations/logo-mark.svg?raw";
import { BRAND, goldMetal, softDisc } from "./kit";
import { parseLogoSvg, toLogoSpace } from "./logoPaths";
import type { Vec2 } from "./math";
import type { LogoState } from "./timeline";

export interface LogoOptions {
  /** Ring tube radius in logo units (the ring's radius is 0.88). */
  ringTube?: number;
  /** Multiplier on the SVG stroke widths: the bake thickens small icons. */
  strokeScale?: number;
  /** Soft additive gold glow behind the mark. */
  halo?: boolean;
}

export interface Logo3D {
  group: Group;
  set(state: LogoState): void;
  dispose(): void;
}

const RING_SEGMENTS = 256;
const RING_RADIAL = 16;

/** A circle in the logo's plane from the top, clockwise (the SVG rotates its stroke by -90°). */
class RingCurve extends Curve<Vector3> {
  constructor(private readonly radius: number) {
    super();
  }

  override getPoint(u: number, target = new Vector3()): Vector3 {
    const a = Math.PI / 2 - u * Math.PI * 2;
    return target.set(Math.cos(a) * this.radius, Math.sin(a) * this.radius, 0);
  }
}

export function createLogo({ ringTube = 0.035, strokeScale = 1, halo = true }: LogoOptions = {}): Logo3D {
  const spec = parseLogoSvg(logoSvg);
  const half = spec.viewBox / 2;
  const ringRadius = spec.ring.r / half;
  const group = new Group();
  const geometries: BufferGeometry[] = [];
  const materials: Material[] = [];

  const ringGeometry = new TubeGeometry(new RingCurve(ringRadius), RING_SEGMENTS, ringTube, RING_RADIAL, true);
  const ringMaterial = goldMetal({ emissive: new Color(BRAND.gold), emissiveIntensity: 0 });
  const ring = new Mesh(ringGeometry, ringMaterial);

  // The SVG's faint full ring under the drawn one, at 18 %.
  const faintGeometry = new TubeGeometry(new RingCurve(ringRadius), 128, ringTube * 0.6, 8, true);
  const faintMaterial = new MeshBasicMaterial({ color: BRAND.gold, transparent: true, opacity: 0, depthWrite: false });
  const faint = new Mesh(faintGeometry, faintMaterial);

  const toV = (p: Vec2) => {
    const [x, y] = toLogoSpace(p, spec);
    return new Vector3(x, y, 0);
  };
  const unit = spec.glyph.scale / half;
  const parts: BufferGeometry[] = [];
  for (const stroke of spec.glyph.strokes) {
    const radius = (stroke.width / 2) * unit * strokeScale;
    for (const sub of stroke.subpaths) {
      const path = new CurvePath<Vector3>();
      let from = sub.start;
      for (const seg of sub.segments) {
        path.add(seg.kind === "L" ? new LineCurve3(toV(from), toV(seg.to)) : new CubicBezierCurve3(toV(from), toV(seg.c1), toV(seg.c2), toV(seg.to)));
        from = seg.to;
      }
      parts.push(new TubeGeometry(path, Math.max(16, sub.segments.length * 48), radius, 12, false));
      // Round caps, like the SVG's stroke-linecap="round".
      if (!sub.closed) {
        for (const end of [sub.start, from]) {
          const v = toV(end);
          parts.push(new SphereGeometry(radius, 12, 8).translate(v.x, v.y, v.z));
        }
      }
    }
  }
  const glyphGeometry = mergeGeometries(parts);
  parts.forEach((part) => part.dispose());
  if (!glyphGeometry) throw new Error("logo: the glyph tubes could not be merged");
  const glyphMaterial = goldMetal({ transparent: true, opacity: 0 });
  const glyph = new Mesh(glyphGeometry, glyphMaterial);

  group.add(faint, ring, glyph);
  geometries.push(ringGeometry, faintGeometry, glyphGeometry);
  materials.push(ringMaterial, faintMaterial, glyphMaterial);

  let haloMaterial: MeshBasicMaterial | null = null;
  if (halo) {
    haloMaterial = new MeshBasicMaterial({ map: softDisc(128), color: BRAND.gold, transparent: true, opacity: 0, blending: AdditiveBlending, depthWrite: false });
    const haloGeometry = new PlaneGeometry(3.2, 3.2);
    const glow = new Mesh(haloGeometry, haloMaterial);
    glow.position.z = -0.3;
    group.add(glow);
    geometries.push(haloGeometry);
    materials.push(haloMaterial);
  }

  return {
    group,
    set({ ring: drawn, glyph: risen, flash }) {
      ringGeometry.setDrawRange(0, Math.floor(drawn * RING_SEGMENTS) * RING_RADIAL * 6);
      ring.visible = drawn > 0;
      ringMaterial.emissiveIntensity = flash * 3; // above the bloom threshold: the one flash when the ring closes
      faintMaterial.opacity = 0.18 * Math.min(1, drawn * 5);
      glyph.visible = risen > 0;
      glyphMaterial.opacity = risen;
      glyph.position.y = (1 - risen) * -0.12; // rises into the ring
      if (haloMaterial) haloMaterial.opacity = 0.35 * drawn;
    },
    dispose() {
      geometries.forEach((g) => g.dispose());
      haloMaterial?.map?.dispose();
      materials.forEach((m) => m.dispose());
    },
  };
}
```

```ts
 * The plant's pasta, modelled in code: spaghetti 1.2, coquillettes, torsadés
 * and vermicelles (seed_client_product_list.py). One instanced mesh per shape,
 * satin in the brand's gold ramp; no model files to download.

import { CatmullRomCurve3, Color, DoubleSide, Euler, InstancedMesh, Matrix4, Quaternion, TubeGeometry, Vector3, type BufferGeometry } from "three";
import { mergeGeometries } from "three/addons/utils/BufferGeometryUtils.js";
import { mixHex } from "./color";
import { BRAND, pastaSatin } from "./kit";
import { SHAPES, type PastaShape, type PieceSeed, type Pose } from "./motion";

const tube = (points: Vector3[], radius: number, tubular: number, radial: number): TubeGeometry =>
  new TubeGeometry(new CatmullRomCurve3(points), tubular, radius, radial, false);

/** Spaghetti 1.2: a long thin rod with a gentle bow, 1.8 long. */
function spaghetti(): BufferGeometry {
  return tube(Array.from({ length: 9 }, (_, i) => { const u = i / 8 - 0.5; return new Vector3(u * 1.8, 0.06 * Math.cos(u * Math.PI), 0); }), 0.022, 48, 8);
}

/** Coquillette: a short open elbow (an arc of 160°); the open ends show it is hollow. */
function coquillette(): BufferGeometry {
  return tube(
    Array.from({ length: 12 }, (_, i) => {
      const a = ((-80 + (160 * i) / 11) * Math.PI) / 180;
      return new Vector3(Math.sin(a) * 0.16, Math.cos(a) * 0.16 - 0.1, 0);
    }),
    0.055,
    24,
    16,
  );
}

/** Torsadé: three strands twisted around one axis, two turns over 0.6. */
function torsade(): BufferGeometry {
  const strands = [0, 1, 2].map((k) =>
    tube(
      Array.from({ length: 40 }, (_, i) => {
        const u = i / 39;
        const a = u * Math.PI * 4 + (k * Math.PI * 2) / 3;
        return new Vector3((u - 0.5) * 0.6, Math.cos(a) * 0.05, Math.sin(a) * 0.05);
      }),
      0.032,
      64,
      8,
    ),
  );
  const merged = mergeGeometries(strands);
  strands.forEach((s) => s.dispose());
  if (!merged) throw new Error("pasta: torsadé strands could not be merged");
  return merged;
}

/** Vermicelle: a fine, short, wavy strand. */
function vermicelle(): BufferGeometry {
  return tube(Array.from({ length: 10 }, (_, i) => { const u = i / 9 - 0.5; return new Vector3(u * 0.5, 0.03 * Math.sin(u * 14), 0.02 * Math.cos(u * 9)); }), 0.012, 32, 6);
}

const BUILDERS: Record<PastaShape, () => BufferGeometry> = { spaghetti, coquillette, torsade, vermicelle };

export interface Pasta {
  meshes: InstancedMesh[];
  update(poses: readonly Pose[]): void;
  dispose(): void;
}

export function createPasta(seeds: readonly PieceSeed[]): Pasta {
  const material = pastaSatin({ side: DoubleSide });
  const colour = new Color();
  const groups = SHAPES.map((shape) => {
    const members = seeds.filter((s) => s.shape === shape);
    const mesh = new InstancedMesh(BUILDERS[shape](), material, Math.max(1, members.length));
    mesh.count = members.length;
    mesh.frustumCulled = false; // instances move every frame; their bounds are never recomputed
    members.forEach((s, j) => mesh.setColorAt(j, colour.set(mixHex(BRAND.gold, BRAND.goldLight, s.tone))));
    return { mesh, members };
  });
  const matrix = new Matrix4();
  const rotation = new Quaternion();
  const euler = new Euler();
  const position = new Vector3();
  const scale = new Vector3();

  return {
    meshes: groups.map((g) => g.mesh),
    update(poses) {
      for (const { mesh, members } of groups) {
        members.forEach((s, j) => {
          const pose = poses[s.index];
          euler.set(pose.rotation[0], pose.rotation[1], pose.rotation[2]);
          matrix.compose(position.set(...pose.position), rotation.setFromEuler(euler), scale.setScalar(pose.scale));
          mesh.setMatrixAt(j, matrix);
        });
        mesh.instanceMatrix.needsUpdate = true;
      }
    },
    dispose() {
      for (const { mesh } of groups) {
        mesh.geometry.dispose();
        mesh.dispose();
      }
      material.dispose();
    },
  };
}
```

```ts
/**
 * Semolina: fine gold grains, plus a few sparks drawn brighter than white so
 * they alone cross the bloom threshold. Positions come from motion.ts each frame.
 */
import { BufferAttribute, BufferGeometry, Color, Points, PointsMaterial } from "three";
import { mixHex } from "./color";
import { BRAND, softDisc } from "./kit";
import type { Vec3 } from "./math";
import { grainPosition, type GrainSeed, type Pose } from "./motion";
import type { IntroState } from "./timeline";

export interface Semolina {
  points: Points[];
  update(time: number, intro: { t: number; state: IntroState } | null, poses: readonly Pose[], pieceCount: number): void;
  dispose(): void;
}

export function createSemolina(seeds: readonly GrainSeed[]): Semolina {
  const sprite = softDisc(64);
  const colour = new Color();
  const layer = (members: readonly GrainSeed[], size: number, brightness: number) => {
    const positions = new Float32Array(members.length * 3);
    const colours = new Float32Array(members.length * 4); // rgba: alpha per grain
    members.forEach((g, i) => {
      colour.set(mixHex(BRAND.goldPellet, BRAND.goldLight, g.tone)).multiplyScalar(brightness);
      colours.set([colour.r, colour.g, colour.b, 0], i * 4);
    });
    const geometry = new BufferGeometry();
    geometry.setAttribute("position", new BufferAttribute(positions, 3));
    geometry.setAttribute("color", new BufferAttribute(colours, 4));
    const material = new PointsMaterial({ size, map: sprite, vertexColors: true, transparent: true, depthWrite: false, sizeAttenuation: true });
    const points = new Points(geometry, material);
    points.frustumCulled = false;
    return { members, positions, colours, geometry, material, points };
  };
  const layers = [layer(seeds.filter((g) => !g.glow), 0.035, 1), layer(seeds.filter((g) => g.glow), 0.09, 2.6)];
  const out: Vec3 = [0, 0, 0];

  return {
    points: layers.map((l) => l.points),
    update(time, intro, poses, pieceCount) {
      for (const l of layers) {
        l.members.forEach((g, i) => {
          const alpha = grainPosition(g, time, intro, poses, pieceCount, out);
          l.positions[i * 3] = out[0];
          l.positions[i * 3 + 1] = out[1];
          l.positions[i * 3 + 2] = out[2];
          l.colours[i * 4 + 3] = alpha;
        });
        l.geometry.attributes.position.needsUpdate = true;
        l.geometry.attributes.color.needsUpdate = true;
      }
    },
    dispose() {
      for (const l of layers) {
        l.geometry.dispose();
        l.material.dispose();
      }
      sprite.dispose();
    },
  };
}
```

```ts
import { createPasta, type Pasta } from "./pasta";
import { TIERS } from "./quality";
import { createSemolina, type Semolina } from "./semolina";
import { cameraPose, cameraPosition, FOV, introState, logoState, withPointer } from "./timeline";

export interface FrameInput {
  /** Stage time in seconds; orbits and drift run on it (the caller slows it to a stop when idle). */
  time: number;
  /** The intro clock while the intro plays, null after it. */
  intro: number | null;
  /** Seconds since the logo appeared in its slot without the intro; Infinity once the intro handed it over. */
  logoBuild: number;
  /** The login's logo slot in CSS px; null keeps the logo centred. */
  slot: Rect | null;
  /** Eased pointer, -1..1 per axis (0 on touch). */
  pointer: { x: number; y: number };
}

export interface WorldOptions {
  tier: "high" | "low";
  transparent?: boolean;
  /** Include the 3D logo (the poster bake leaves it out: the login slots show it). */
  logo?: boolean;
  preserve?: boolean;
}

export interface World {
  frame(input: FrameInput): void;
  setTier(tier: "high" | "low"): void;
  resize(width: number, height: number, pixelRatio: number): void;
  dispose(): void;
}

const SEED = 20261002;

interface Layers {
  count: number;
  pasta: Pasta;
  semolina: Semolina;
}

export function createWorld(canvas: HTMLCanvasElement, { tier, transparent = false, logo: withLogo = true, preserve = false }: WorldOptions): World {
  const stage = createStage(canvas, { transparent, bloom: TIERS[tier].bloom, preserve });
  const scene = new Scene();
  const navy = untoneColor(BRAND.navyDeep); // lands on #071F43 after tone mapping: the canvas edge is the CSS navy
  if (!transparent) scene.background = navy;
  scene.fog = new Fog(navy, 6, 26);
  scene.environment = stage.environment;
  scene.environmentIntensity = ENV_INTENSITY;
  scene.add(new HemisphereLight(0xffffff, BRAND.navyDeep, 0.45));
  const key = new DirectionalLight(0xffffff, 1.8);
  key.position.set(-4, 8, 6);
  scene.add(key);

  const camera = new PerspectiveCamera(FOV, 1, 0.05, 200);
  const pieces = pieceSeeds(TIERS.high.pieces, SEED);
  const grains = grainSeeds(TIERS.high.grains, SEED + 1);
  const poses: Pose[] = pieces.map(() => ({ position: [0, 0, 0], rotation: [0, 0, 0], scale: 0 }));

  const build = (next: "high" | "low"): Layers => {
    const settings = TIERS[next];
    const layers = {
      count: settings.pieces,
      pasta: createPasta(pieces.slice(0, settings.pieces)),
      semolina: createSemolina(grains.slice(0, settings.grains)),
    };
    scene.add(...layers.pasta.meshes, ...layers.semolina.points);
    stage.setBloom(settings.bloom);
    return layers;
  };
  const clear = (old: Layers) => {
    scene.remove(...old.pasta.meshes, ...old.semolina.points);
    old.pasta.dispose();
    old.semolina.dispose();
  };
  let layers = build(tier);

  const logo = withLogo ? createLogo() : null;
  if (logo) scene.add(logo.group);

  let width = 1;
  let height = 1;
  const slotPoint = new Vector3();

  return {
    frame({ time, intro, logoBuild, slot, pointer }) {
      const state = intro === null ? null : introState(intro);
      const pose = withPointer(cameraPose(intro, time, width / height), pointer);
      camera.position.set(...cameraPosition(pose));
      camera.lookAt(pose.target[0], pose.target[1], pose.target[2]);
      camera.updateMatrixWorld();

      for (let i = 0; i < layers.count; i++) poses[i] = piecePose(pieces[i], time, state);
      layers.pasta.update(poses);
      layers.semolina.update(time, intro !== null && state ? { t: intro, state } : null, poses, layers.count);

      if (logo) {
        logo.set(logoState(intro, logoBuild));
        // Centred during the intro; glued to its login slot after the handoff (the slot is read every frame).
        let size = 1;
        slotPoint.set(0, 0, 0);
        if (slot) {
          const placed = placeInSlot(slot, width, height, FOV, LOGO_SLOT_DEPTH);
          const u = state ? state.handoff : 1;
          const target = camera.localToWorld(new Vector3(placed.x, placed.y, placed.z));
          slotPoint.set(lerp(0, target.x, u), lerp(0, target.y, u), lerp(0, target.z, u));
          size = lerp(1, placed.size / 2, u); // the mark spans 2 world units at scale 1
        }
        logo.group.position.copy(slotPoint);
        logo.group.scale.setScalar(size);
        logo.group.rotation.set(-0.12 + 0.05 * Math.sin(time * 0.37), 0.22 * Math.sin(time * 0.5), 0);
      }
      stage.render(scene, camera);
    },
    setTier(next) {
      clear(layers);
      layers = build(next);
    },
    resize(w, h, pixelRatio) {
      width = w;
      height = h;
      camera.aspect = w / h;
      camera.updateProjectionMatrix();
      stage.setSize(w, h, pixelRatio);
    },
    dispose() {
      clear(layers);
      logo?.dispose();
      stage.dispose();
    },
  };
}
```

```ts
/**
 * Dev-only bake page (frontend/bake-brand.html): renders the 3D logo images and
 * the fallback posters from the same code as the live scene, so they always
 * match it. `node scripts/bake-brand.mjs` (dev server up) saves them in place.
 * Never in the production build: Vite builds index.html only.
 */
import { DirectionalLight, HemisphereLight, PerspectiveCamera, Scene } from "three";
import { BRAND, createStage, ENV_INTENSITY } from "./kit";
import { createLogo, type LogoOptions } from "./logo";
import { createWorld } from "./world";

export interface BakeOutput {
  name: string;
  dataUrl: string;
}

// Ring and strokes thicken as the image shrinks, so the mark still reads at 16–32 px.
const THIN: LogoOptions = { ringTube: 0.035, strokeScale: 1 };
const MEDIUM: LogoOptions = { ringTube: 0.05, strokeScale: 1.2 };
const THICK: LogoOptions = { ringTube: 0.07, strokeScale: 1.5 };
const MARK_PX = 512;

function canvas2d(size: number) {
  const c = document.createElement("canvas");
  c.width = c.height = size;
  const ctx = c.getContext("2d")!;
  ctx.imageSmoothingQuality = "high";
  return { c, ctx };
}

/** The gold mark alone on a transparent 512 px square, lit like the live scene. */
function renderMark(options: LogoOptions): HTMLCanvasElement {
  const canvas = document.createElement("canvas");
  const stage = createStage(canvas, { transparent: true, bloom: false, preserve: true });
  stage.setSize(MARK_PX, MARK_PX, 1);
  const scene = new Scene();
  scene.environment = stage.environment;
  scene.environmentIntensity = ENV_INTENSITY;
  scene.add(new HemisphereLight(0xffffff, BRAND.navyDeep, 0.5));
  const key = new DirectionalLight(0xffffff, 2);
  key.position.set(-3, 5, 6);
  scene.add(key);
  const logo = createLogo({ ...options, halo: false });
  logo.set({ ring: 1, glyph: 1, flash: 0 });
  logo.group.rotation.set(-0.1, -0.28, 0);
  scene.add(logo.group);
  const camera = new PerspectiveCamera(15, 1, 0.1, 100);
  camera.position.set(0, 0, 1 / (0.92 * Math.tan((15 * Math.PI) / 360))); // the mark (radius ~1) fills 92 %
  camera.lookAt(0, 0, 0);
  stage.render(scene, camera);
  const { c, ctx } = canvas2d(MARK_PX);
  ctx.drawImage(canvas, 0, 0);
  logo.dispose();
  stage.dispose();
  return c;
}

/** Halve until near the target, then draw at size: smoother than one big jump. */
function downscale(src: HTMLCanvasElement, size: number): HTMLCanvasElement {
  let current = src;
  while (current.width / 2 >= size) {
    const { c, ctx } = canvas2d(current.width / 2);
    ctx.drawImage(current, 0, 0, c.width, c.height);
    current = c;
  }
  const { c, ctx } = canvas2d(size);
  ctx.drawImage(current, 0, 0, size, size);
  return c;
}

/** The mark on a navy rounded square with a soft gold glow (the login panel's look). `radius` and `padding` are shares of the side. */
function tile(mark: HTMLCanvasElement, size: number, radius: number, padding: number): HTMLCanvasElement {
  const { c, ctx } = canvas2d(size);
  ctx.beginPath();
  ctx.roundRect(0, 0, size, size, radius * size);
  ctx.fillStyle = BRAND.navyDeep;
  ctx.fill();
  const glow = ctx.createRadialGradient(size / 2, size / 2, 0, size / 2, size / 2, size / 2);
  glow.addColorStop(0, "rgba(245,169,0,0.22)");
  glow.addColorStop(1, "rgba(245,169,0,0)");
  ctx.fillStyle = glow;
  ctx.fill();
  const inner = Math.round(size * (1 - 2 * padding));
  const offset = Math.round((size - inner) / 2);
  ctx.drawImage(downscale(mark, inner), offset, offset);
  return c;
}

/** The login scene without the logo (the slots show it), as the fallback poster. */
function poster(width: number, height: number): string {
  const canvas = document.createElement("canvas");
  const world = createWorld(canvas, { tier: "high", logo: false, preserve: true });
  world.resize(width, height, 1);
  world.frame({ time: 14, intro: null, logoBuild: Infinity, slot: null, pointer: { x: 0, y: 0 } });
  const url = canvas.toDataURL("image/webp", 0.82);
  world.dispose();
  return url;
}

function bake(): BakeOutput[] {
  const outputs: BakeOutput[] = [];
  const png = (name: string, c: HTMLCanvasElement) => outputs.push({ name, dataUrl: c.toDataURL("image/png") });
  png("logo-3d-mark-512.png", renderMark(THIN));
  const medium = renderMark(MEDIUM);
  for (const size of [64, 96, 88, 132]) png(`logo-3d-${size}.png`, tile(medium, size, 0.28, 0.14));
  const thick = renderMark(THICK);
  for (const size of [16, 32]) png(`favicon-${size}.png`, tile(thick, size, 0.22, 0.08));
  png("apple-touch-icon.png", tile(medium, 180, 0, 0.16)); // iOS rounds the corners itself
  outputs.push({ name: "sonalim-stage-landscape.webp", dataUrl: poster(1920, 1080) });
  outputs.push({ name: "sonalim-stage-portrait.webp", dataUrl: poster(1080, 1920) });
  return outputs;
}

declare global {
  interface Window {
    __brandBake?: Promise<BakeOutput[]>;
  }
}

window.__brandBake = Promise.resolve().then(() => {
  const outputs = bake();
  const list = document.getElementById("out")!;
  for (const { name, dataUrl } of outputs) {
    const figure = document.createElement("figure");
    const img = document.createElement("img");
    img.src = dataUrl;
    img.alt = name;
    const caption = document.createElement("figcaption");
    caption.textContent = name;
    figure.append(img, caption);
    list.appendChild(figure);
  }
  return outputs;
});
```

```js
// Minimal Chrome DevTools Protocol client for the brand scripts: a private
// headless Chrome on its own port and profile (other sessions share the
// Playwright browser), driven over the WebSocket that Node 22+ has built in.
import { spawn } from "node:child_process";
import { mkdtempSync, rmSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";

export const CHROME = process.env.CHROME ?? "C:/Program Files/Google/Chrome/Application/chrome.exe";

export const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

export async function launchChrome({ port = 9344, flags = [] } = {}) {
  const profile = mkdtempSync(join(tmpdir(), "sonalim-brand-"));
  const proc = spawn(
    CHROME,
    [
      "--headless=new",
      `--remote-debugging-port=${port}`,
      `--user-data-dir=${profile}`,
      "--use-angle=swiftshader",
      "--enable-unsafe-swiftshader",
      "--ignore-gpu-blocklist",
      "--hide-scrollbars",
      ...flags,
      "about:blank",
    ],
    { stdio: "ignore" },
  );
  let wsUrl = null;
  for (let i = 0; i < 50 && !wsUrl; i++) {
    try {
      const targets = await (await fetch(`http://127.0.0.1:${port}/json/list`)).json();
      wsUrl = targets.find((t) => t.type === "page")?.webSocketDebuggerUrl ?? null;
    } catch {
      // Chrome is not listening yet.
    }
    if (!wsUrl) await sleep(200);
  }
  if (!wsUrl) {
    proc.kill();
    throw new Error(`Chrome did not open a page on port ${port} (CHROME=${CHROME})`);
  }
  const ws = new WebSocket(wsUrl);
  await new Promise((resolve, reject) => {
    ws.onopen = resolve;
    ws.onerror = reject;
  });
  let id = 0;
  const pending = new Map();
  ws.onmessage = (event) => {
    const msg = JSON.parse(event.data);
    const waiter = pending.get(msg.id);
    if (!waiter) return;
    pending.delete(msg.id);
    if (msg.error) waiter.reject(new Error(msg.error.message));
    else waiter.resolve(msg.result);
  };
  const send = (method, params = {}) =>
    new Promise((resolve, reject) => {
      const n = ++id;
      pending.set(n, { resolve, reject });
      ws.send(JSON.stringify({ id: n, method, params }));
    });
  /** Evaluates `expression` in the page, awaiting a promise, and returns its JSON value. */
  const evaluate = async (expression) => {
    const { result, exceptionDetails } = await send("Runtime.evaluate", { expression, awaitPromise: true, returnByValue: true });
    if (exceptionDetails) throw new Error(exceptionDetails.exception?.description ?? exceptionDetails.text);
    return result.value;
  };
  /** Re-evaluates `expression` until it is truthy (a navigation may replace the page in between). */
  const waitFor = async (expression, timeoutMs = 60000) => {
    const end = Date.now() + timeoutMs;
    for (;;) {
      try {
        const value = await evaluate(expression);
        if (value) return value;
      } catch {
        // The page is still loading.
      }
      if (Date.now() > end) throw new Error(`timed out waiting for: ${expression}`);
      await sleep(200);
    }
  };
  const close = () => {
    ws.close();
    proc.kill();
    try {
      rmSync(profile, { recursive: true, force: true });
    } catch {
      // Chrome may still hold the profile for a moment on Windows.
    }
  };
  return { send, evaluate, waitFor, close };
}
```

```tsx
 * 2026-10-02 §3–5). Mounted once in App. It renders only while the intro plays
 * or the login page is mounted, shows a baked poster when there is no live 3D,

import posterLandscape from "../../../assets/animations/sonalim-stage-landscape.webp";
import posterPortrait from "../../../assets/animations/sonalim-stage-portrait.webp";

import { stageStore, stageWanted, useStage } from "./stageStore";

const IDLE_STOP_MS = 60_000;

function liveSupported(): boolean {
  if (brandFlags.fallback) return false;
  if (window.matchMedia?.("(prefers-reduced-motion: reduce)").matches) return false;
  try {

/** The login's first visible logo slot, in CSS px (a slot hidden at this width measures 0 and is skipped). */
function slotRect(): Rect | null {
  for (const el of stageStore.get().loginSlots ?? []) {
    const r = el.getBoundingClientRect();
    if (r.width > 0 && r.height > 0) return { left: r.left, top: r.top, width: r.width, height: r.height };
  }
  return null;
}

/** Dev only: lets scripts/brand-stills.mjs report which tier ran. */

export function BrandStage() {
  const wanted = useStage(stageWanted);
  const fallback = useStage((s) => s.fallback);
  const ready = useStage((s) => s.ready);
  const phase = useStage((s) => s.introPhase);
  const introStarted = useStage((s) => s.introStarted);
  const containerRef = useRef<HTMLDivElement>(null);
  // Keep the layer one fade longer than `wanted`, so leaving the login fades the navy out.
  const [shown, setShown] = useState(wanted);
  // The poster stays under a fading-in canvas until the fade is over.
  const [settled, setSettled] = useState(false);

  useEffect(() => {
    if (wanted) {
      setShown(true);
      return;
    }
    const id = window.setTimeout(() => setShown(false), 300);
    return () => window.clearTimeout(id);
  }, [wanted]);

  useEffect(() => {
    if (!ready) {
      setSettled(false);
      return;
    }
    const id = window.setTimeout(() => setSettled(true), FADE_MS);
    return () => window.clearTimeout(id);
  }, [ready]);

  useEffect(() => {
    if (!wanted || fallback) return;
    if (!liveSupported()) {
      markTier("fallback");
      stageStore.set({ fallback: true });
      return;
    }
    const container = containerRef.current;
    if (!container) return;
    // A fresh canvas per run: a disposed renderer loses its context, and a lost canvas cannot be reused.
    const canvas = document.createElement("canvas");
    canvas.className = "absolute inset-0 h-full w-full";
    canvas.style.zIndex = "1";
    canvas.style.opacity = "0";
    canvas.style.transition = `opacity ${FADE_MS}ms ease-out`;
    container.appendChild(canvas);

    let world: World | null = null;
    let disposed = false;
    let raf = 0;
    let tier: Tier = brandFlags.tier ?? "high";
    let samples: number[] = [];
    let warmup = 5;
    let last = 0;
    let time = 0;
    let motion = 1;
    let lastInput = performance.now();
    const pointer = { x: 0, y: 0 };
    const target = { x: 0, y: 0 };
    let handedOff = false;
    let buildStart: number | null = null;

    const goFallback = () => {
      markTier("fallback");
      stageStore.set({ fallback: true });
    };

    const frame = (now: number) => {
      if (!world) return;
      const dt = last ? Math.min(0.1, (now - last) / 1000) : 0; // a stall never jumps the orbits
      last = now;
      const s = stageStore.get();
      const introRunning = s.introPhase === "playing" || s.introPhase === "handoff";
      if (s.introPhase === "handoff") handedOff = true;
      if (introRunning) stageStore.startIntro(); // the intro's clock starts on its first live frame
      const idle = !introRunning && now - lastInput > IDLE_STOP_MS;
      motion += ((idle ? 0 : 1) - motion) * Math.min(1, dt * 2);
      time += dt * motion;
      const ease = Math.min(1, dt * 3);
      pointer.x += (target.x - pointer.x) * ease;
      pointer.y += (target.y - pointer.y) * ease;
      if (!introRunning && !handedOff && buildStart === null) buildStart = now;
      world.frame({
        time,
        intro: introRunning ? stageStore.introTime() : null,
        logoBuild: handedOff ? Infinity : buildStart === null ? 0 : (now - buildStart) / 1000,
        slot: slotRect(),
        pointer,
      });
      if (!s.ready) {
        canvas.style.opacity = "1";
        stageStore.set({ ready: true });
      }
      if (brandFlags.tier || tier === "still") return;
      if (warmup > 0) {
        warmup--;
        return;
      }
      if (dt > 0) samples.push(dt * 1000);
      if (samples.length < SAMPLE_FRAMES) return;
      const next = nextTier(tier, samples, !introRunning);
      samples = [];
      if (next === tier) return;
      tier = next;
      markTier(tier);
      if (tier !== "still") {
        world.setTier(tier);
        resize();
      }
    };

    const loop = () => {
      raf = requestAnimationFrame((now) => {
        raf = 0;
        frame(now);
        // Stop when frozen, hidden, or at rest after the idle ease (input restarts it).
        if (!disposed && tier !== "still" && !document.hidden && motion > 0.001) loop();
      });
    };

    function resize() {
      if (!world) return;
      world.resize(window.innerWidth, window.innerHeight, pixelRatio(window.devicePixelRatio || 1, tier));
      if (!raf) frame(performance.now()); // a resting or frozen scene still follows the window
    }

    const wake = () => {
      lastInput = performance.now();
      if (!raf && world && !disposed && tier !== "still" && !document.hidden) loop();
    };
    const onPointerMove = (event: PointerEvent) => {
      wake();
      if (event.pointerType !== "mouse") return;
      target.x = (event.clientX / window.innerWidth) * 2 - 1;
      target.y = (event.clientY / window.innerHeight) * 2 - 1;
    };
    const onVisibility = () => {
      if (document.hidden) {
        cancelAnimationFrame(raf);
        raf = 0;
      } else {
        last = 0;
        wake();
      }
    };
    const onLost = (event: Event) => {
      event.preventDefault();
      goFallback();
    };

    window.addEventListener("resize", resize);
    window.addEventListener("pointermove", onPointerMove);
    window.addEventListener("pointerdown", wake);
    window.addEventListener("keydown", wake);
    document.addEventListener("visibilitychange", onVisibility);
    canvas.addEventListener("webglcontextlost", onLost);
    markTier(tier);

    import("./world").then(
      ({ createWorld }) => {
        if (disposed) return;
        try {
          world = createWorld(canvas, { tier: tier === "still" ? "low" : tier });
        } catch {
          goFallback();
          return;
        }
        resize();
        loop();
      },
      () => {
        if (!disposed) goFallback();
      },
    );

    return () => {
      disposed = true;
      cancelAnimationFrame(raf);
      window.removeEventListener("resize", resize);
      window.removeEventListener("pointermove", onPointerMove);
      window.removeEventListener("pointerdown", wake);
      window.removeEventListener("keydown", wake);
      document.removeEventListener("visibilitychange", onVisibility);
      // Before dispose: forcing the context loss on purpose is not a failure.
      canvas.removeEventListener("webglcontextlost", onLost);
      world?.dispose();
      canvas.remove();
      stageStore.set({ ready: false });
    };
  }, [wanted, fallback]);

  if (!wanted && !shown) return null;

  const introRunning = phase === "playing" || phase === "handoff";
  // During the intro: plain navy until the clock runs, then the poster only if the live 3D is not there yet.
  const showPoster = fallback || (introRunning ? introStarted && !ready : !settled);

  return (
    <div
      ref={containerRef}
      aria-hidden="true"
      className="pointer-events-none fixed inset-0 z-0 transition-opacity duration-300"
      style={{ opacity: wanted ? 1 : 0, background: NAVY_GLOW }}
    >
      {showPoster && (
        <picture>
          <source media="(orientation: portrait)" srcSet={posterPortrait} />
          <img src={posterLandscape} alt="" className="absolute inset-0 h-full w-full object-cover" />
        </picture>
      )}
    </div>
  );
}
```

```tsx
const POSTER_WAIT_MS = 1500;

/** Wait for the web font so the wordmark doesn't swap mid-animation, but never let a slow font host hold the splash. */
function fontsReady(): Promise<unknown> {
  const fonts =
    typeof document !== "undefined" && "fonts" in document
      ? document.fonts.ready
      : Promise.resolve();
  const cap = new Promise<void>((resolve) =>
    window.setTimeout(resolve, FONT_WAIT_CAP_MS),

 * a session was restored: no session = the 3D welcome intro (about 4 s,

    };

        onFinished?.();

/**

  const started = useStage((s) => s.introStarted);

    if (fallback) stageStore.startIntro();

  useEffect(() => {
    stageStore.set({ introPhase: "playing" });

      stageStore.freezeIntro(brandFlags.introAt);
```

```tsx
  hidden: { opacity: 0, y: 14 },
  show: { opacity: 1, y: 0, transition: { duration: 0.45, ease: EASE } },
};

/**
 * Navy behind the brand panel's text: at least 85 % up to 65 % of the panel's
 * height, so white text keeps 7:1 or better whatever pasta passes behind (spec §3).
 * The top of the panel stays clear for the live 3D logo.
 */
const PANEL_SCRIM = "linear-gradient(to top, rgba(7,31,67,0.9) 0%, rgba(7,31,67,0.85) 80%, rgba(7,31,67,0) 100%)";

const fieldClass =
  "h-11 rounded-xl border-slate-200 bg-slate-50/80 pl-10 text-slate-900 shadow-sm transition-all duration-200 placeholder:text-slate-400 focus:border-[#F5A900] focus:bg-white focus-visible:ring-4 focus-visible:ring-[#F5A900]/15";

export function LoginPage() {
  const { login } = useAuth();
  const { t, language, setLanguage } = useI18n();
  const navigate = useNavigate();
  const reduceMotion = useReducedMotion();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [pending, setPending] = useState(false);
  const panelSlotRef = useRef<HTMLDivElement>(null);
  const heroSlotRef = useRef<HTMLDivElement>(null);
  const revealed = useStage(loginRevealed);
  // The live 3D logo flies into the slots; until it draws (or with no live 3D) the baked mark holds the place.
  const live = useStage((s) => s.ready && !s.fallback);

  // Before paint, so the stage's navy is under the page from its first frame (no white flash).
  useLayoutEffect(() => {
    const slots = [panelSlotRef.current, heroSlotRef.current].filter((el): el is HTMLDivElement => el !== null);
    stageStore.set({ loginSlots: slots });
    return () => stageStore.set({ loginSlots: null });
  }, []);

  async function handleSubmit(event: FormEvent) {
    event.preventDefault();
    setError(null);
    setPending(true);
    try {
      await login(email, password);
      navigate("/crm", { replace: true });
    } catch (cause) {
      setError(cause instanceof ApiError ? cause.message : t("login.failed"));
    } finally {
      setPending(false);
    }
  }

  return (
    <div className="theme-login relative z-10 flex min-h-screen flex-col items-center justify-center overflow-hidden px-4 py-8">
      {/* Phones have no brand panel: the live logo hovers here, above the card. */}
      <div ref={heroSlotRef} aria-hidden="true" className="mb-6 h-24 w-24 md:hidden">
        {revealed && !live && <img src={logoMark} alt="" className="h-full w-full" />}
      </div>

      <motion.div
        variants={card}
        initial="hidden"
        animate={revealed ? "show" : "hidden"}
        className="relative w-full max-w-5xl overflow-hidden rounded-[32px] border border-white/15 shadow-[0_35px_90px_rgba(0,0,0,0.45)]"
      >
        <div className="grid grid-cols-1 md:grid-cols-[1.1fr_0.9fr]">
          {/* Brand panel: a window onto the 3D scene. */}
          <div className="relative hidden min-h-[580px] flex-col justify-between overflow-hidden border-r border-white/10 p-8 text-white md:flex">
            <div aria-hidden="true" className="pointer-events-none absolute inset-x-0 bottom-0 top-[18%]" style={{ background: PANEL_SCRIM }} />

            <motion.div variants={fromLeft} className="relative z-10 flex items-center gap-4">
              <div ref={panelSlotRef} aria-hidden="true" className="h-20 w-20 shrink-0">
                {!live && <img src={logoMark} alt="" className="h-full w-full" />}
              </div>
              <div className="rounded-xl bg-[#071F43]/85 px-3 py-2">
                <BrandWordmark tone="dark" size={22} />
              </div>
            </motion.div>

            <motion.div variants={fromBelow} className="relative z-10">
              <p className="text-xs uppercase tracking-[0.35em] text-[#F5A900]/90">{t("login.tagline")}</p>
              <h1 className="mt-5 max-w-md text-4xl font-bold leading-tight text-white">{t("login.heroTitle")}</h1>
              <p className="mt-4 max-w-md text-base text-slate-300">{t("login.heroSubtitle")}</p>
            </motion.div>

            <motion.ul variants={stagger} className="relative z-10 grid gap-3">
              {FEATURES.map((key) => (
                <motion.li
                  key={key}
                  variants={rise}
                  whileHover={reduceMotion ? undefined : { x: 6 }}
                  className="flex items-center gap-3 rounded-2xl border border-white/10 bg-white/5 px-4 py-3 text-sm text-slate-100 transition-colors hover:border-[#F5A900]/40 hover:bg-white/10"
                >
                  <span className="inline-flex h-2.5 w-2.5 shrink-0 rounded-full bg-[#F5A900] shadow-[0_0_16px_rgba(245,169,0,0.9)]" />
                  {t(key)}
                </motion.li>
              ))}
            </motion.ul>
          </div>

          {/* Form panel: near-opaque white, no backdrop blur over the live canvas. */}
          <motion.div variants={stagger} className="flex flex-col bg-white/95 p-6 sm:p-8 md:p-10">
            <motion.div variants={rise} className="mb-8 flex items-center justify-between gap-3">
              <div className="flex items-center gap-3">
                <img
                  src={logoTile88}
                  srcSet={`${logoTile88} 2x, ${logoTile132} 3x`}
                  alt=""
                  width={44}
                  height={44}
                  className="h-11 w-11 shrink-0 rounded-xl shadow-[0_10px_30px_rgba(245,169,0,0.35)]"
                />
                <div>
                  <BrandWordmark tone="light" size={15} />
                  <p className="mt-0.5 flex items-center gap-1 text-xs text-muted-foreground">
                    <Lock className="h-3 w-3" aria-hidden="true" />
                    {t("login.secureAccess")}
                  </p>
                </div>
              </div>
              <div className="flex items-center gap-2">
                <span className="sr-only">{t("login.language")}</span>
                <LanguageToggle language={language} onChange={setLanguage} layoutId="login-lang-toggle" />
              </div>
            </motion.div>

            <motion.div variants={rise} className="mb-6">
              <h2 className="text-[1.8rem] font-semibold leading-tight text-slate-900">{t("login.title")}</h2>
              <p className="mt-2 text-sm text-slate-500">{t("login.subtitle")}</p>
            </motion.div>

            <form className="space-y-4" onSubmit={handleSubmit}>
              <motion.div variants={rise} className="space-y-1.5">
                <Label htmlFor="email" className="text-sm font-medium text-slate-700">
                  {t("login.email")}
                </Label>
                <div className="relative">
                  <Mail className="pointer-events-none absolute left-3.5 top-1/2 z-10 h-4 w-4 -translate-y-1/2 text-slate-400" aria-hidden="true" />
                  <Input
                    id="email"
                    name="email"
                    type="email"
                    autoComplete="username"
                    autoFocus
                    required
                    placeholder={t("login.emailPlaceholder")}
                    value={email}
                    onChange={(event) => setEmail(event.target.value)}
                    className={fieldClass}
                  />
                </div>
              </motion.div>

              <motion.div variants={rise} className="space-y-1.5">
                <Label htmlFor="password" className="text-sm font-medium text-slate-700">
                  {t("login.password")}
                </Label>
                <div className="relative">
                  <Lock className="pointer-events-none absolute left-3.5 top-1/2 z-10 h-4 w-4 -translate-y-1/2 text-slate-400" aria-hidden="true" />
                  <Input
                    id="password"
                    name="password"
                    type={showPassword ? "text" : "password"}
                    autoComplete="current-password"
                    required
                    value={password}
                    onChange={(event) => setPassword(event.target.value)}
                    className={`${fieldClass} pr-11`}
                  />
                  <button
                    type="button"
                    onClick={() => setShowPassword((value) => !value)}
                    aria-label={showPassword ? t("login.hidePassword") : t("login.showPassword")}
                    aria-pressed={showPassword}
                    className="absolute right-2 top-1/2 z-10 flex h-8 w-8 -translate-y-1/2 items-center justify-center rounded-lg text-slate-400 transition-colors hover:bg-slate-100 hover:text-slate-700 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#F5A900]/40"
                  >
                    {showPassword ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
                  </button>
                </div>
              </motion.div>

              {error && (
                <motion.p
                  role="alert"
                  initial={{ opacity: 0, y: -6 }}
                  animate={{ opacity: 1, y: 0 }}
                  className="animate-shake rounded-lg border border-destructive/20 bg-destructive/5 px-3 py-2 text-sm text-destructive"
                >
                  {error}
                </motion.p>
              )}

              <motion.div variants={rise} className="pt-1">
                <motion.div whileHover={{ scale: pending || reduceMotion ? 1 : 1.01 }} whileTap={{ scale: pending || reduceMotion ? 1 : 0.99 }}>
                  <Button
                    type="submit"
                    className="h-11 w-full rounded-xl bg-gradient-to-r from-primary to-[#1D4F8C] text-base text-white shadow-[0_12px_28px_rgba(18,59,109,0.35)] transition hover:from-[#0E3160] hover:to-primary hover:shadow-[0_16px_32px_rgba(18,59,109,0.45)] disabled:opacity-80"
                    disabled={pending}
                  >
                    {pending && <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />}
                    {pending ? t("login.submitting") : t("login.submit")}
                  </Button>
                </motion.div>
              </motion.div>
            </form>

            <motion.p variants={rise} className="mt-auto pt-8 text-center text-xs text-slate-400">
              {t("login.footer")}
            </motion.p>
          </motion.div>
        </div>
      </motion.div>
    </div>
  );
}
```
